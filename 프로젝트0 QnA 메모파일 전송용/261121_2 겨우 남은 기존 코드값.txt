    function switchTab(type) {
      // 1. 모든 탭 숨기기
      const allTabs = ['meetings', 'todos', 'ideas', 'resources'];
      allTabs.forEach(t => {
        const el = document.getElementById('tab-' + t);
        if (el) el.style.display = 'none';
      });

      // 2. 모든 버튼 비활성화
      const allBtns = ['btn-m', 'btn-t', 'btn-i', 'btn-r'];
      allBtns.forEach(b => {

        if (el) el.classList.remove('active');
      });

      const targetTab = document.getElementById('tab-' + type);
      if (targetTab) targetTab.style.display = 'block';

      const btnMap = { 'todos': 'btn-t', 'meetings': 'btn-m', 'ideas': 'btn-i', 'resources': 'btn-r' };

  let editingTodoId = null;

  async function loadTodos() {
    const list = document.getElementById('todos-list');
    if (!list) return;
    const { data: todos, error } = await supabase.from('todos').select('*').order('created_at', { ascending: false });

    if (error) {
      console.error(error);
      return;
    }

    if (!todos || todos.length === 0) {
      list.innerHTML = '<div style="text-align:center; padding:20px; color:#aaa;">등록된 작업이 없습니다. 오늘 할 일을 추가해 보세요! 🚀</div>';
      return;
    }

    list.innerHTML = todos.map(todo => {
      // 수정 모드일 때 카드 자리에서 바로 열리는 입력창
      if (editingTodoId === todo.id) {
        return `

            <button onclick="saveEditTodo(${todo.id})" class="pink-btn" style="padding: 6px 12px; font-size: 0.8rem; border-radius: 6px;">저장</button>
            <button onclick="cancelEditTodo()" style="padding: 6px 12px; font-size: 0.8rem; border: 1px solid #ccc; background: white; border-radius: 6px; cursor: pointer;">취소</button>
          </div>
        `;
      }

      // 일반 모드 (체크박스 + 작업명 + 수정/삭제 버튼)
      return `
        <div class="card" style="display: flex; justify-content: space-between; align-items: center; padding: 12px 16px; margin-bottom: 8px; background: white; border-radius: 10px; border: 1.5px solid #ffe4e9;">
          <div style="display: flex; align-items: center; gap: 10px; flex: 1;">

            </span>

            <button onclick="deleteTodo(${todo.id})" style="background: none; border: none; color: #ff8fa3; cursor: pointer; font-size: 0.85rem; font-weight: bold;">삭제</button>
          </div>
        </div>
      `;
    }).join('');
  }

  function startEditTodo(id) {
    editingTodoId = id;
    loadTodos();
  }

  function cancelEditTodo() {
    editingTodoId = null;
    loadTodos();
  }

  async function saveEditTodo(id) {
    const input = document.getElementById(`edit-todo-task-${id}`);
    const newTask = input.value.trim();
    if (!newTask) {
      alert('작업 내용을 입력해주세요!');
      return;
    }
    await supabase.from('todos').update({ task: newTask }).eq('id', id);
    editingTodoId = null;
    loadTodos();
  }

  async function addTodo() {
    const input = document.getElementById('todo-input');
    const task = input.value.trim();
    if (!task) return;
    await supabase.from('todos').insert([{ task: task, is_completed: false }]);
    input.value = '';
    loadTodos();
  }

  function handleTodoKey(event) {
    if (event.key === 'Enter') addTodo();
  }

  async function toggleTodo(id, status) {
    await supabase.from('todos').update({ is_completed: status }).eq('id', id);
    loadTodos();
  }

  async function deleteTodo(id) {
    if (!confirm('이 작업을 삭제할까요?')) return;
    await supabase.from('todos').delete().eq('id', id);
    loadTodos();
  }

  function startEditTodo(id) {

  function addIdea() {
    const titleInput = document.getElementById('idea-title');
    const descInput = document.getElementById('idea-desc');
    const title = titleInput.value.trim();
    const desc = descInput.value.trim();

    if (!title) {
      alert('아이디어 제목을 입력해주세요!');
      return;
    }

    const now = new Date();
    const dateStr = `${now.getFullYear()}.${now.getMonth()+1}.${now.getDate()} ${now.getHours()}:${String(now.getMinutes()).padStart(2, '0')}`;
    
    ideas.unshift({ title, desc, date: dateStr });
    localStorage.setItem('p0_ideas', JSON.stringify(ideas));
    titleInput.value = '';
    descInput.value = '';
    renderIdeas();
  }

    if (!title || !url) {
      alert('자료 이름과 링크 URL을 모두 입력해주세요!');
      return;

    localStorage.setItem('p0_resources', JSON.stringify(resources));
    titleInput.value = '';

  // 보안 글자 변환
  function escapeHtml(str) {
    if (!str) return '';
    return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
  }

  window.switchTab = function(type) {
    const allTabs = ['meetings', 'todos', 'ideas', 'resources'];
    allTabs.forEach(t => {
      const el = document.getElementById('tab-' + t);
      if (el) el.style.display = 'none';
    });

    const allBtns = ['btn-m', 'btn-t', 'btn-i', 'btn-r'];
    allBtns.forEach(b => {
      const el = document.getElementById(b);
      if (el) el.classList.remove('active');
    });

    const targetTab = document.getElementById('tab-' + type);
    if (targetTab) targetTab.style.display = 'block';

    const btnMap = { 'todos': 'btn-t', 'meetings': 'btn-m', 'ideas': 'btn-i', 'resources': 'btn-r' };
    const targetBtn = document.getElementById(btnMap[type]);
    if (targetBtn) targetBtn.classList.add('active');
  };
